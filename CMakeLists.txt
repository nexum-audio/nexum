cmake_minimum_required(VERSION 3.22)
project(NexumAudio VERSION 0.12.0)
add_subdirectory(External/JUCE JUCE)
juce_add_binary_data(SynthBrand SOURCES Assets/NEXUM.png Assets/NEXUM_SYNTH.png Assets/NEXUM_FX.png)
file(GLOB ZSTD_COMMON External/zstd/lib/common/*.c)
file(GLOB ZSTD_DECOMPRESS External/zstd/lib/decompress/*.c)
add_library(SerumZstd STATIC ${ZSTD_COMMON} ${ZSTD_DECOMPRESS})
target_include_directories(SerumZstd PUBLIC External/zstd/lib)
target_compile_definitions(SerumZstd PRIVATE ZSTD_DISABLE_ASM=1 ZSTD_LEGACY_SUPPORT=0)
# One canonical source list: both products always build the same FX algorithms,
# parameter definitions, library storage and rack UI. No copied FX implementation.
set(NEXUM_SHARED_SOURCES Source/Processor.cpp Source/Editor.cpp Source/ExternalPresetImport.cpp
 Source/Expansion.cpp Source/PresetLibrary.cpp Source/PackBrowser.cpp Source/FxRack.cpp
 Source/Sequencer.cpp Source/LiveModulation.cpp)
function(nexum_common target)
 target_sources(${target} PRIVATE ${NEXUM_SHARED_SOURCES})
 target_compile_features(${target} PRIVATE cxx_std_17)
 target_include_directories(${target} PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/shared/NexumDesign/include")
 target_compile_definitions(${target} PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
 target_link_libraries(${target} PRIVATE SynthBrand SerumZstd juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags)
 if(MSVC)
  target_compile_options(${target} PRIVATE /utf-8 /MP1)
 endif()
endfunction()
